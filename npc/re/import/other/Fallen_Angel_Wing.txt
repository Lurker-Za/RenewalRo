								specialeffect2 EF_REPAIRWEAPON;

								goto suend;
								close;
							case 4712:
							case 4713:
							case 4714:
								if(.@enchan1_rand == 1) set .@equip_card[1],4713; //INT+4

							case 4722:
							case 4723:
							case 4724:
								if(.@enchan1_rand == 1) set .@equip_card[1],4723; //DEX+4

								goto suend;
								close;
							case 4752:
							case 4753:
							case 4754:
								if(.@enchan1_rand == 1) set .@equip_card[1],4753; //LUK+4
								if(.@enchan1_rand == 2) set .@equip_card[1],4753; //LUK+4

						}
					case 2:
						mes .@npcname$;
						mes "취소하셨습니다.";
						close;
					}
				}
			}
			break;
		case 2: // 초기화
			mes .@npcname$;
			mes "^0000ff" + .@itemname$ + "^000000 인챈트 초기화를 위해선",

	end;
