-	script	Mercenary Merchant#dummy::MercMerchant	-1,{
	mes "[물품 판매상]";
	mes "안녕하세요.";
	mes "용병에 관한 물품을 판매하고 있습니다.";
	mes "뭘 찾으시나요?";
	next;
	setarray .@item, 12184,	12185,	12241,	12242,	12243;
	setarray .@cost, 2500,	5000,	800,	1500,	3000;
	for (.@size = getarraysize(.@item); .@i < .@size; .@i++)
		.@menu$ = .@menu$ + getitemname(.@item[.@i]) + ":";
	.@m = select(.@menu$)-1;
	.@str$ = .@cost[.@m] + "";
	.@len = getstrlen(.@str$);
	.@str_cost$ = (.@len <= 3)? .@str$ : insertchar(.@str$,",",.@len-3);
	mes "[물품 판매상]";
	mes "'" + getitemname(.@item[.@m]) + "' 이신가요?";
	mes "개당 " + .@str_cost$ + "z 에 판매하고 있습니다.";
	mes "얼마나 필요하신가요?";
	next;
	input .@input;
	if (set(.@total_po, .@input * .@cost[.@m]) > Zeny) {
		mes "[물품 판매상]";
		mes "가지고 계신 제니가 부족하시군요.";
		mes "가지고 계신 제니량 내에서 정해주세요.";
		close;
	}
	if (!checkweight(.@item[.@m], .@input)) {
		mes "[물품 판매상]";
		mes "주문하신 물품의 무게가 소지가능한 무게를 초과하네요.";
		mes "무게를 비우신 후 다시 주문해주세요.";
		close;
	}
	mes "[물품 판매상]";
	mes "이용해주셔서 감사합니다.";
	mes "주문하신 물품은 '" + getitemname(.@item[.@m]) + "'";
	mes .@input + "개 입니다.";
	if(.@input > 0)
		Zeny = Zeny - .@total_po;
		getitem .@item[.@m], .@input;
	close;
}

//============================================================
-	script	Mercenary_Switch	-1,{

	setarray .@name$, "Spear","Sword","Bow";
	.@npc$ = strnpcinfo(2);
	for (.@size = getarraysize(.@name$); .@type < .@size; .@type++)
		if (.@npc$ == .@name$[.@type]) break;

	mes "[Checker]";
	mes "Please input a password.";
	next;
	.@i = callfunc("F_GM_NPC",1854,0,0,10000);
	mes "[Checker]";
	if (.@i == -2)
		mes "Error.";
	else if (.@i == 0)
		mes "Wrong number";
	else if (.@i == 1) {
		mes "Please select.";
		next;
		switch( select( "Turn off Mercenary NPC","Turn on Mercenary NPC" ) ) {
		case 1:
			mes "NPCs are turned off.";
			disablenpc "Mercenary Manager#" + .@name$[.@type];
			disablenpc "Mercenary Merchant#" + .@name$[.@type];
			break;
		case 2:
			mes "NPCs are turned on.";
			enablenpc "Mercenary Manager#" + .@name$[.@type];
			enablenpc "Mercenary Merchant#" + .@name$[.@type];
			break;
		}
	}
	close;
}
